cmake_minimum_required(VERSION 3.20)
project(noxtls_cc13xx_full_tests C)
enable_testing()
get_filename_component(NOXTLS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NOXTLS_PROFILE crypto_only CACHE STRING "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_APPLICATIONS OFF CACHE BOOL "" FORCE)
set(NOXTLS_OMIT_UT_SOURCES ON CACHE BOOL "" FORCE)
set(NOXTLS_CFG_FEATURE_CC13XX_AES_ACCEL ON CACHE BOOL "" FORCE)
set(NOXTLS_CFG_FEATURE_CC13XX_P256_ACCEL ON CACHE BOOL "" FORCE)
set(NOXTLS_CFG_FEATURE_CC13XX_X25519_ACCEL ON CACHE BOOL "" FORCE)
set(NOXTLS_CFG_FEATURE_X25519 ON CACHE BOOL "" FORCE)
set(NOXTLS_ERROR_UTNOX "" CACHE PATH "Existing UTNox checkout")
add_subdirectory("${NOXTLS_ERROR_UTNOX}" utnox)
add_subdirectory("${NOXTLS_ROOT}" noxtls)
add_executable(noxtls_cc13xx_full_ut cc13xx_full_ut.c)
target_link_libraries(noxtls_cc13xx_full_ut PRIVATE utnox noxtls_encryption noxtls_hash noxtls_pkc)
target_include_directories(noxtls_cc13xx_full_ut PRIVATE "${NOXTLS_ROOT}"
    "${NOXTLS_ROOT}/noxtls-lib/common" "${NOXTLS_ROOT}/noxtls-lib")
set_target_properties(noxtls_cc13xx_full_ut PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
add_test(NAME noxtls_cc13xx_full_ut COMMAND noxtls_cc13xx_full_ut RUN_ALL)
