if(NOT TARGET noxtls_tls OR NOT NOXTLS_CFG_FEATURE_TLS13)
    message(FATAL_ERROR "Handshake tests require TLS13")
endif()
if(NOT TARGET utnox)
    set(NOXTLS_ERROR_UTNOX "" CACHE PATH "Existing UTNox checkout")
    if(NOT EXISTS "${NOXTLS_ERROR_UTNOX}/CMakeLists.txt")
        message(FATAL_ERROR "Set NOXTLS_ERROR_UTNOX to an existing UTNox checkout")
    endif()
    add_subdirectory("${NOXTLS_ERROR_UTNOX}" utnox)
endif()
add_executable(tls13_link_ut tls13_link_ut.c)
target_link_libraries(tls13_link_ut PRIVATE noxtls_tls utnox)
add_test(NAME tls13_link_ut COMMAND tls13_link_ut RUN_ALL)
if(NOXTLS_CFG_FEATURE_DTLS AND NOXTLS_CFG_FEATURE_X25519 AND NOXTLS_CFG_FEATURE_AES_GCM)
    add_executable(dtls13_epoch_policy_ut dtls13_epoch_policy_ut.c)
    target_link_libraries(dtls13_epoch_policy_ut PRIVATE noxtls_tls utnox)
    set_target_properties(dtls13_epoch_policy_ut PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    add_test(NAME dtls13_epoch_policy_ut COMMAND dtls13_epoch_policy_ut RUN_ALL)
    add_executable(dtls13_recovery_ut dtls13_recovery_ut.c)
    target_link_libraries(dtls13_recovery_ut PRIVATE noxtls_tls utnox)
    set_target_properties(dtls13_recovery_ut PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    add_test(NAME dtls13_recovery_ut COMMAND dtls13_recovery_ut RUN_ALL)
else()
    message(STATUS "Skipping loss tests: require DTLS, X25519 and AES_GCM; retaining TLS link check")
endif()
